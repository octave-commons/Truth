```clojure
;; --- Mass/CoM propagation ---------------------------------------------------

(defn- node-mass [node] (if node (double (:mass node)) 0.0))
(defn- node-com  [node] (if node (:com node) (sp/vec3 0.0 0.0 0.0)))

(defn propagate-mass
  "Walk tree bottom-up, computing total mass and center of mass for
   every internal node from its children.
   Returns a new tree with :mass and :com correct at every level."
  [node]
  (cond
    (nil? node) nil

    (leaf-node? node) node

    (internal-node? node)
    (let [children' (mapv propagate-mass (:children node))
          total-mass (reduce #(+ %1 (node-mass %2)) 0.0 children')
          com (if (zero? total-mass)
                (sp/center (:aabb node))
                (->> children'
                     (reduce (fn [acc child]
                               (sp/v+ acc (sp/v* (node-com child)
                                                 (node-mass child))))
                             (sp/vec3 0.0 0.0 0.0))
                     (#(sp/v* % (/ 1.0 total-mass)))))]
      (assoc node
             :children children'
             :mass     total-mass
             :com      com))))

(defn build-tree
  "Build a Barnes–Hut octree from a seq of Body records.
   Returns a fully propagated tree or nil for empty input."
  [bodies]
  (when (seq bodies)
    (let [bb   (-> (bounding-aabb-for-bodies bodies)
                   ;; pad slightly so bodies on the boundary classify cleanly
                   (update :min #(sp/v+ % [-1e-6 -1e-6 -1e-6]))
                   (update :max #(sp/v+ % [1e-6 1e-6 1e-6])))
          tree (reduce insert-body-into-node nil bodies)]
      (propagate-mass tree))))

;; --- Acceleration evaluation ------------------------------------------------

(def ^:const default-theta 0.5)
(def ^:const softening     1e-4) ;; softening length in AU (or whatever units)

(defn- accel-from-mass
  "Gravitational acceleration on a test body at position `pos`
   due to aggregate mass `mass` at center-of-mass `com`, with
   gravitational constant G and softening ε.
   Returns vec3 acceleration."
  [^double G pos mass com]
  (let [r     (sp/v- com pos)
        r2    (+ (sp/len2 r) (* softening softening))
        r3    (* r2 (Math/sqrt r2))
        scale (/ (* G (double mass)) r3)]
    (sp/v* r scale)))

(defn- traverse
  "Recursive Barnes–Hut traversal.
   Accumulates acceleration on body at `pos` from `node`."
  [^double G ^double theta pos acc node]
  (cond
    (nil? node) acc

    (leaf-node? node)
    (let [body (:body node)]
      ;; skip self-interaction
      (if (= (:id body) (:id (meta pos)))
        acc
        (sp/v+ acc (accel-from-mass G pos (:mass body) (:position body)))))

    (internal-node? node)
    (let [s (sp/max-side (:aabb node))
          d (sp/dist pos (:com node))]
      (if (< (/ s d) theta)
        ;; treat entire node as point mass
        (sp/v+ acc (accel-from-mass G pos (:mass node) (:com node)))
        ;; recurse into children
        (reduce (fn [a child] (traverse G theta pos a child))
                acc
                (:children node))))))

(defn acceleration
  "Compute gravitational acceleration on `body` from all bodies in `tree`.
   G     — gravitational constant (units consistent with your simulation)
   theta — Barnes–Hut opening angle (default 0.5); smaller = more accurate
   Returns vec3 acceleration."
  ([G tree body]
   (acceleration G default-theta tree body))
  ([G theta tree body]
   (let [;; tag pos with body id so leaf skips self
         pos (with-meta (:position body) {:id (:id body)})]
     (traverse G theta pos (sp/vec3 0.0 0.0 0.0) tree))))
```

This is the complete Barnes–Hut evaluation. The $s/d < \theta$ criterion is the classic approximation condition from the original Barnes \& Hut 1986 paper. At $\theta = 0.5$, error is typically under 1% for cosmological simulations; tightening to $\theta = 0.3$ gives sub-0.1% for orbital mechanics where precision matters more.

## `domain.orbital.integrator`: Leapfrog Verlet

Leapfrog is the correct integrator here. It is symplectic — it conserves a shadow Hamiltonian exactly, so energy drift over long timescales is bounded rather than accumulating. Euler blows up planetary orbits over millions of ticks; RK4 is accurate but not symplectic; Leapfrog splits the difference perfectly for orbital mechanics.

The scheme is: given positions $\mathbf{x}_n$, half-step velocities $\mathbf{v}_{n+1/2}$:

$$
\mathbf{v}_{n+1/2} = \mathbf{v}_{n-1/2} + \mathbf{a}(\mathbf{x}_n) \cdot \Delta t
$$

$$
\mathbf{x}_{n+1} = \mathbf{x}_n + \mathbf{v}_{n+1/2} \cdot \Delta t
$$

```clojure
(ns domain.orbital.integrator
  "Symplectic Leapfrog (Störmer–Verlet) integrator for n-body orbital mechanics.
   All positions and velocities are vec3 from shape.spatial.
   Requires a pure acceleration function (fn [body] -> vec3)."
  (:require
    [shape.spatial :as sp]))

;; --- Single body step -------------------------------------------------------

(defn leapfrog-kick
  "Velocity half-step (kick):
   v_half = v + a * (dt/2)"
  [body accel-fn ^double dt]
  (let [a (accel-fn body)]
    (update body :velocity sp/v+ (sp/v* a (* dt 0.5)))))

(defn leapfrog-drift
  "Position full-step (drift):
   x_new = x + v * dt"
  [body ^double dt]
  (update body :position sp/v+ (sp/v* (:velocity body) dt)))

(defn leapfrog-step
  "Full Leapfrog step (kick-drift-kick):
   1. kick velocity by dt/2 using acceleration at current position
   2. drift position by full dt using new velocity
   3. kick velocity by dt/2 using acceleration at new position
   Returns updated body."
  [body accel-fn ^double dt]
  (-> body
      (leapfrog-kick accel-fn dt)
      (leapfrog-drift dt)
      (leapfrog-kick accel-fn dt)))

;; --- N-body batch step ------------------------------------------------------

(defn step-all
  "Advance all bodies by one Leapfrog step.
   accel-fn-for: (fn [tree body] -> vec3 acceleration)
   Builds a fresh Barnes–Hut tree once per tick, then evaluates
   acceleration per body. Pure — returns updated seq of bodies."
  [bodies build-tree-fn accel-fn-for ^double dt]
  (let [tree (build-tree-fn bodies)]
    (mapv (fn [body]
            (leapfrog-step body
                           (fn [b] (accel-fn-for tree b))
                           dt))
          bodies)))
```

## `domain.orbital.kepler`: Two-body Kepler solution (fast path for dominated systems)

For systems with one dominant star and small satellites, full Barnes–Hut every tick is wasteful. The fast path is a direct Kepler solver. This also gives you a reference solution for validating Barnes–Hut accuracy.

```clojure
(ns domain.orbital.kepler
  "Two-body Kepler orbit utilities.
   - kepler-period: orbital period from semi-major axis + GM
   - mean-anomaly: M(t) from epoch
   - eccentric-anomaly: E from M via Newton iteration
   - true-anomaly: ν from E
   - orbital-state: position + velocity from orbital elements at time t"
  (:require [shape.spatial :as sp]))

(def ^:const two-pi (* 2.0 Math/PI))

(defn kepler-period
  "Orbital period T = 2π √(a³/GM).
   a  — semi-major axis (AU or consistent units)
   GM — gravitational parameter (G * M_primary)"
  [^double a ^double GM]
  (* two-pi (Math/sqrt (/ (* a a a) GM))))

(defn mean-anomaly
  "Mean anomaly M(t) = 2π(t - t0)/T, wrapped to [0, 2π)."
  [^double t ^double t0 ^double T]
  (mod (* two-pi (/ (- t t0) T)) two-pi))

(defn eccentric-anomaly
  "Solve Kepler's equation M = E - e*sin(E) by Newton–Raphson.
   Converges in ~5 iterations for e < 0.99."
  ([^double M ^double e]
   (eccentric-anomaly M e 1e-10 50))
  ([^double M ^double e ^double tol ^long max-iter]
   (loop [E M, i 0]
     (let [dE (/ (- E (* e (Math/sin E)) M)
                 (- 1.0 (* e (Math/cos E))))]
       (cond
         (< (Math/abs dE) tol) E
         (>= i max-iter)
         (throw (ex-info "eccentric-anomaly: no convergence"
                         {:M M :e e :E E :i i}))
         :else (recur (- E dE) (inc i)))))))

(defn true-anomaly
  "True anomaly ν from eccentric anomaly E and eccentricity e."
  [^double E ^double e]
  (* 2.0 (Math/atan2
           (* (Math/sqrt (+ 1.0 e)) (Math/sin (* E 0.5)))
           (* (Math/sqrt (- 1.0 e)) (Math/cos (* E 0.5))))))

(defn orbital-state
  "Compute position (vec3) and velocity (vec3) in the orbital plane
   from classic elements at time t.
   Elements map keys:
     :a  — semi-major axis
     :e  — eccentricity
     :i  — inclination (rad)
     :Ω  — longitude of ascending node (rad)
     :ω  — argument of periapsis (rad)
     :t0 — epoch (time of periapsis passage)
     :GM — gravitational parameter of primary"
  [{:keys [^double a ^double e ^double i
           ^double Ω ^double ω ^double t0 ^double GM]}
   ^double t]
  (let [T   (kepler-period a GM)
        M   (mean-anomaly t t0 T)
        E   (eccentric-anomaly M e)
        ν   (true-anomaly E e)
        ;; distance from focus
        r   (/ (* a (- 1.0 (* e e)))
               (+ 1.0 (* e (Math/cos ν))))
        ;; position in perifocal frame
        px  (* r (Math/cos ν))
        py  (* r (Math/sin ν))
        ;; velocity in perifocal frame: h = √(GM·a·(1-e²))
        h   (Math/sqrt (* GM a (- 1.0 (* e e))))
        vx  (/ (* (- (Math/sin ν)) GM) h)
        vy  (/ (* (+ e (Math/cos ν)) GM) h)
        ;; Rotation matrices: ω → i → Ω
        cΩ  (Math/cos Ω) sΩ (Math/sin Ω)
        cω  (Math/cos ω) sω (Math/sin ω)
        ci  (Math/cos i)  si (Math/sin i)
        ;; Perifocal → inertial (standard 3-1-3 rotation)
        Rxx (- (* cΩ cω) (* sΩ sω ci))
        Rxy (- (- (* cΩ sω)) (* sΩ cω ci))
        Ryx (+ (* sΩ cω) (* cΩ sω ci))
        Ryy (+ (- (* sΩ sω)) (* cΩ cω ci))
        Rzx (* sω si)
        Rzy (* cω si)]
    {:position [(+ (* Rxx px) (* Rxy py))
                (+ (* Ryx px) (* Ryy py))
                (+ (* Rzx px) (* Rzy py))]
     :velocity [(+ (* Rxx vx) (* Rxy vy))
                (+ (* Ryx vx) (* Ryy vy))
                (+ (* Rzx vx) (* Rzy vy))]}))
```

## `domain.orbital.system`: the tick loop

Now we can wire everything together into a clean tick that maps the LOD zones from the architecture brief — immediate (every frame), regional (every N ticks), global (every M ticks):

```clojure
(ns domain.orbital.system
  "Top-level orbital simulation tick.
   Manages LOD zones: immediate / regional / global.
   Exposes a single `tick` function over immutable world-state."
  (:require
    [domain.gravity.barnes-hut :as bh]
    [domain.orbital.integrator :as integrator]
    [shape.spatial :as sp]))

;; --- LOD constants ----------------------------------------------------------

(def ^:long immediate-every  1)
(def ^:long regional-every  60)   ;; ~1 s at 60 fps
(def ^:long global-every  3600)   ;; ~60 s at 60 fps

;; --- World state shape ------------------------------------------------------
;; world = {:tick       long
;;          :dt         double    (seconds per tick)
;;          :G          double    (gravitational constant in sim units)
;;          :theta      double    (Barnes–Hut θ)
;;          :bodies     [Body]
;;          :regional   any       (coarse aggregate, opaque here)
;;          :global     any}

(defn empty-world
  "Construct an empty world with defaults."
  [{:keys [dt G theta]
    :or   {dt 1.0 G 6.674e-11 theta 0.5}}]
  {:tick    0
   :dt      dt
   :G       G
   :theta   theta
   :bodies  []
   :regional nil
   :global   nil})

(defn add-body
  "Pure: add a Body to the world."
  [world body]
  (update world :bodies conj body))

;; --- Tick subphases ---------------------------------------------------------

(defn- tick-immediate
  "Run full orbital physics for all bodies in the immediate zone.
   Returns updated world with new :bodies."
  [{:keys [G theta dt bodies] :as world}]
  (let [tree     (bh/build-tree bodies)
        accel-fn (fn [tree body]
                   (bh/acceleration G theta tree body))
        bodies'  (integrator/step-all
                   bodies
                   (constantly tree)        ;; tree already built
                   accel-fn
                   dt)]
    ;; Re-use the already-built tree: pass identity build fn
    ;; Actually step-all rebuilds internally; bypass by inlining:
    (assoc world :bodies
           (mapv (fn [b]
                   (integrator/leapfrog-step
                     b
                     (fn [b'] (bh/acceleration G theta tree b'))
                     dt))
                 bodies))))

(defn- tick-global

;; --- Main tick --------------------------------------------------------------

(defn tick
  "Advance world by one simulation step.
   LOD gates:
   - immediate: every tick (orbital physics)
   - regional:  every regional-every ticks (coarse weather/hydrology)
   - global:    every global-every ticks (civilization epoch)"
  [{:keys [tick] :as world}]
  (let [n (long tick)]
    (cond-> (tick-immediate world)
      (zero? (mod n regional-every)) tick-regional
      (zero? (mod n global-every))   tick-global
      true                           (update :tick inc))))
```

```clojure
    (is (= 5.0 (double (:mass tree))))

      (is (< (Math/abs cz) 1e-9)))))

(deftest kepler-earth-period
  (let [AU   1.495978707e11   ;; meters

        days (/ T 86400.0)]
    (is (< (Math/abs (- days 365.25)) 0.1)
        (str "Expected ~365.25 days, got " days))))

;; μ-integrator-1: energy conservation over circular orbit (Leapfrog, 1 year, 100 steps)
;; Checks that |ΔE/E| < 1e-4 — hallmark of symplectic integrator
(deftest leapfrog-energy-conservation
  (let [GM   1.327124e20
        r    1.495978707e11
        v_c  (Math/sqrt (/ GM r))   ;; circular orbital speed
        body {:id :earth :mass 5.972e24 :radius 6.371e6 :kind :planet
              :position [r 0.0 0.0] :velocity [0.0 v_c 0.0]}
        ;; simple 1-body: accel toward origin with GM
        accel (fn [b]
                (let [pos (:position b)
                      r2  (sp/len2 pos)
                      r3  (* r2 (Math/sqrt r2))]
                  (sp/v* pos (/ (- GM) r3))))

        rel   (Math/abs (/ (- ef e0) (Math/abs e0)))]

        (str "Energy drift " rel " exceeds threshold 1e-4"))))
```
