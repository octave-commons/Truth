## `shape.core`: Shapes, Claims, Events, ids

This module is pure category: no validation, no IO, just data constructors and id helpers.

```clojure
(ns shape.core
  "Core category-level shapes for the engine:
   - Shape: describes form/kind, never validity
   - Claim: assertion that a value instantiates a Shape in some context
   - Event: immutable occurrence built from Claims
   - simple id helpers for resources and shapes."
  (:import
    (java.util UUID)))

;; --- Id helpers -------------------------------------------------------------

(defn new-shape-id
  "Allocate a new opaque identifier for a Shape.
   Keep it a bare UUID for now; wrap later if needed."
  []
  (UUID/randomUUID))

(defn new-claim-id
  "Allocate a new opaque identifier for a Claim."
  []
  (UUID/randomUUID))

(defn new-resource-id
  "Allocate a new opaque identifier for any resource (contract, registry, etc.)."
  []
  (UUID/randomUUID))

;; --- Records ----------------------------------------------------------------

(defrecord Shape
  [id           ;; UUID
   kind         ;; e.g. :state | :transition | :relation | domain-specific keyword
   form         ;; description of structure (schema/type/blueprint)
   name         ;; human-readable name (not necessarily unique)
   description] ;; human-readable description
  )

(defn ->shape
  "Construct a Shape from a map.
   Required keys:
   - :id   (UUID)
   - :kind (keyword)
   - :form (schema/blueprint)
   Optional:
   - :name
   - :description"
  [{:keys [id kind form name description] :as m}]
  (when-not (and id kind form)
    (throw (ex-info "Shape requires :id, :kind, and :form"
                    {:kind ::invalid-shape :shape m})))
  (->Shape id kind form name description))

(defrecord Claim
  [id          ;; UUID
   shape-id    ;; UUID of Shape this claim instantiates
   value       ;; concrete value
   context     ;; map: {:tick int ...} or richer context
   asserted-by ;; keyword or resource id indicating who made the claim
   ])

(defn ->claim
  "Construct a Claim tying a Shape to a concrete value in a context.
   Required keys:
   - :id
   - :shape-id
   - :value
   Optional:
   - :context
   - :asserted-by"
  [{:keys [id shape-id value context asserted-by] :as m}]
  (when-not (and id shape-id (contains? m :value))
    (throw (ex-info "Claim requires :id, :shape-id, and :value"
                    {:kind ::invalid-claim :claim m})))
  (->Claim id shape-id value context asserted-by))

(defrecord Event
  [id         ;; UUID
   timestamp  ;; monotonically increasing logical or wall-clock time
   kind       ;; keyword, e.g. :component-updated :entity-created
   cause      ;; who/what caused this (system id, external source, prior event id)
   claims]    ;; vector of Claim
  )

(defn new-event-id
  []
  (UUID/randomUUID))

(defn ->event
  "Construct an Event. Claims are treated as already-validated; higher layers
   decide whether the event is admissible under current laws."
  [{:keys [id timestamp kind cause claims] :as m}]
  (when-not (and timestamp kind (sequential? claims))
    (throw (ex-info "Event requires :timestamp, :kind, and :claims (sequential)"
                    {:kind ::invalid-event :event m})))
  (->Event (or id (new-event-id)) timestamp kind cause (vec claims)))
```

This is intentionally minimal: no Malli, no Typed Clojure yet. Those live in laws.

## `law.contract`: Contracts over Shapes \& Claims

This module defines what it means for a Contract to validate a Claim, with the `:type` vs `:quality` distinction you described:

- `:type` ⇒ no extra keys beyond the schema.
- `:quality` ⇒ extra keys allowed, only explicit constraints enforced.

For now, we treat `:schema` as a simple “map of key → predicate” so the μ‑specs are executable without pulling Malli. You can later flip this to Malli while preserving the public API.

```clojure
(ns law.contract
  "Contracts govern which Claims over Shapes are admissible.
   They do not perform side-effects; they return facts about validity
   that other layers can record in the ledger."
  (:require
    [shape.core :as shape]))

(defrecord Contract
  [id          ;; UUID (resource id)
   shape-id    ;; UUID of the Shape being governed
   kind        ;; :type | :quality | other domain-specific kinds
   schema      ;; description used for validation (for now: {k predicate})
   name        ;; human-readable name
   description ;; human-readable description
   ;; optional hooks
   on-true     ;; (fn [state claim] state')
   on-false    ;; (fn [state claim violation] state')
   on-any])    ;; (fn [state claim result] state')
  )

(defn ->contract
  "Construct a Contract. Required keys:
   - :id
   - :shape-id
   - :kind       (e.g. :type or :quality)
   - :schema     (validation description)
   Optional:
   - :name
   - :description
   - :on-true, :on-false, :on-any (pure state transformers)"
  [{:keys [id shape-id kind schema name description on-true on-false on-any] :as m}]
  (when-not (and id shape-id kind schema)
    (throw (ex-info "Contract requires :id, :shape-id, :kind, and :schema"
                    {:kind ::invalid-contract :contract m})))
  (->Contract id shape-id kind schema name description on-true on-false on-any))

;; --- Validation -------------------------------------------------------------

(def ^:const ok ::ok)
(def ^:const violation ::violation)

(defn- value-keys
  "Utility: safe read of top-level keys for a claim value; non-maps return empty set."
  [v]
  (if (map? v) (set (keys v)) #{}))

(defn- validate-schema
  "Validate `value` against simple predicate schema:
   schema = {k predicate}, all predicates must return truthy.
   Returns nil on success, or a seq of {:path [k] :value v :reason keyword}."
  [schema value]
  (reduce
    (fn [acc [k pred]]
      (let [v (get value k ::missing)]
        (cond
          (= v ::missing)
          (conj acc {:path   [k]
                     :value  nil
                     :reason :missing})

          (not (pred v))
          (conj acc {:path   [k]
                     :value  v
                     :reason :invalid})

          :else acc)))
    []
    schema))

(defn- type-extra-keys
  "For :type contracts, extra keys beyond schema are violations."
  [schema value]
  (let [allowed (set (keys schema))
        actual  (value-keys value)
        extra   (clojure.set/difference actual allowed)]
    (map (fn [k]
           {:path   [k]
            :value  (get value k)
            :reason :unexpected-key})
         extra)))

(defn validate
  "Validate a Claim against a Contract.
   Returns either:
   - ::ok
   - {:result ::violation
      :contract contract
      :claim    claim
      :issues   vector-of-issue-maps}

   For :type contracts:
   - value must satisfy all schema predicates
   - no extra keys beyond those in schema

   For :quality contracts:
   - value must satisfy all schema predicates
   - extra keys are allowed"
  [^Contract c ^shape.core.Claim claim]
  (let [{:keys [kind schema]} c
        value       (:value claim)
        base-issues (validate-schema schema value)
        extra-issues (if (= kind :type)
                       (type-extra-keys schema value)
                       [])
        issues      (into base-issues extra-issues)]
    (if (seq issues)
      {:result   violation
       :contract c
       :claim    claim
       :issues   issues}
      ok)))
```

This satisfies the μ‑spec behaviors:

- Type vs quality semantics.
- `::ok` vs a structured violation map.
- No direct side effects; ledger recording is a separate concern.

## `law.registry`: Lawful, typed registries

Registry is your “no junk drawer” index: a typed, contract‑governed collection with dense storage and id indexing. This matches the “simple but explicit” approach in lightweight Clojure ECS libs, while giving you a proper contract layer on top.[^5_1]

```clojure
(ns law.registry
  "Registry for resources of a single type, governed by a Contract.
   Guarantees:
   - every item satisfies resource-contract
   - ids are unique
   - index is consistent with items"
  (:require
    [law.contract :as contract]))

(defrecord Registry
  [resource-contract  ;; Contract
   items              ;; vector of resources
   index])            ;; {id -> idx}
  )

(defn ->registry
  "Create an empty Registry for resources governed by `resource-contract`."
  [resource-contract]
  (->Registry resource-contract [] {}))

(defn- ensure-unique-id!
  [index id resource]
  (when (contains? index id)
    (throw (ex-info "Registry duplicate id"
                    {:kind    ::duplicate-id
                     :id      id
                     :attempt resource}))))

(defn- validate-resource!
  [resource-contract resource]
  ;; We expect resources to have an :id key at minimum.
  (let [id (:id resource)]
    (when (nil? id)
      (throw (ex-info "Registry resource missing :id"
                      {:kind     ::missing-id
                       :resource resource})))
    ;; Adapt resource -> pseudo-claim for validation.
    (let [claim {:id       id
                 :shape-id (:shape-id resource-contract)
                 :value    resource}
          res   (contract/validate resource-contract claim)]
      (when (not= contract/ok res)
        (throw (ex-info "Registry contract violation"
                        {:kind     ::contract-violation
                         :issues   (:issues res)
                         :resource resource}))))))

(defn add
  "Add a resource to the registry.
   - Validates the resource against the registry's contract.
   - Enforces unique id.
   - Returns a new Registry."
  [^Registry reg resource]
  (let [{:keys [resource-contract items index]} reg
        id (:id resource)]
    (validate-resource! resource-contract resource)
    (ensure-unique-id! index id resource)
    (let [idx (count items)]
      (->Registry
        resource-contract
        (conj items resource)
        (assoc index id idx)))))

(defn get-by-id
  "Lookup a resource by id. Returns nil if not present."
  [^Registry reg id]
  (let [{:keys [items index]} reg
        idx (get index id)]
    (when (some? idx)
      (nth items idx))))

(defn rebuild-index
  "Rebuild the id -> idx index from the items vector.
   Useful as a law check or repair logic if index is suspected corrupt."
  [^Registry reg]
  (let [{:keys [resource-contract items]} reg
        new-index (into {}
                        (map-indexed
                          (fn [idx {:keys [id]}]
                            (when (nil? id)
                              (throw (ex-info "Registry item missing :id while rebuilding index"
                                              {:kind  ::missing-id
                                               :index idx})))
                            [id idx]))
                        items)]
    (->Registry resource-contract items new-index)))
```

This satisfies μ1:

- Validates every added resource via `resource-contract`.
- Enforces unique `:id`.
- Consistent `items`/`index` with `rebuild-index`.
