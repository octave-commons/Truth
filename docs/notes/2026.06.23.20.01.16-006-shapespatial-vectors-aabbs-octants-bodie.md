## `shape.spatial`: vectors, AABBs, octants, bodies

```clojure
(ns shape.spatial
  "Domain-agnostic 3D spatial primitives:
   - vec3 arithmetic
   - AABB (axis-aligned bounding boxes)
   - octant classification
   - Body record (mass, radius, position, velocity, kind).")

;; --- vec3 -------------------------------------------------------------------

(defn vec3
  (^clojure.lang.IPersistentVector [x y z] [double x double y double z])
  (^clojure.lang.IPersistentVector [^double x ^double y] [x y 0.0])
  (^clojure.lang.IPersistentVector [^double x] [x 0.0 0.0])
  (^clojure.lang.IPersistentVector [] [0.0 0.0 0.0]))

(defn v+
  "Componentwise addition of two vec3s."
  [[ax ay az] [bx by bz]]
  [(+ (double ax) (double bx))
   (+ (double ay) (double by))
   (+ (double az) (double bz))])

(defn v-
  "Componentwise subtraction: a - b."
  [[ax ay az] [bx by bz]]
  [(- (double ax) (double bx))
   (- (double ay) (double by))
   (- (double az) (double bz))])

(defn v*
  "Scale vec3 by scalar s."
  [[ax ay az] s]
  (let [s (double s)]
    [(* ax s) (* ay s) (* az s)]))

(defn dot
  "Dot product of two vec3s."
  [[ax ay az] [bx by bz]]
  (+ (* (double ax) (double bx))
     (* (double ay) (double by))
     (* (double az) (double bz))))

(defn len2
  "Squared length of vec3."
  [v]
  (dot v v))

(defn len
  "Euclidean length of vec3."
  [v]
  (Math/sqrt (len2 v)))

(defn dist
  "Distance between two vec3s."
  [a b]
  (len (v- b a)))

;; --- AABB -------------------------------------------------------------------

(defrecord AABB
  [min ;; vec3
   max ;; vec3
   ])

(defn aabb
  "Construct an AABB from min and max vec3s."
  [min max]
  (->AABB min max))

(defn aabb-from-points
  "Smallest AABB containing all given points (vec3s)."
  [points]
  (assert (seq points) "aabb-from-points: need at least one point")
  (let [[[x0 y0 z0] & more] points]
    (loop [min-x (double x0)
           min-y (double y0)
           min-z (double z0)
           max-x (double x0)
           max-y (double y0)
           max-z (double z0)
           ps   more]
      (if-let [[x y z] (first ps)]
        (recur (min min-x (double x))
               (min min-y (double y))
               (min min-z (double z))
               (max max-x (double x))
               (max max-y (double y))
               (max max-z (double z))
               (next ps))
        (->AABB [min-x min-y min-z] [max-x max-y max-z])))))

(defn aabb-include
  "Extend AABB to include point p."
  [^AABB bb [x y z]]
  (let [[min-x min-y min-z] (:min bb)
        [max-x max-y max-z] (:max bb)]
    (->AABB [(min min-x (double x))
             (min min-y (double y))
             (min min-z (double z))]
            [(max max-x (double x))
             (max max-y (double y))
             (max max-z (double z))])))

(defn contains?
  "Does the AABB contain point p (inclusive)?"
  [^AABB bb [x y z]]
  (let [[min-x min-y min-z] (:min bb)
        [max-x max-y max-z] (:max bb)
        x (double x) y (double y) z (double z)]
    (and (<= min-x x max-x)
         (<= min-y y max-y)
         (<= min-z z max-z))))

(defn center
  "Center of the AABB."
  [^AABB bb]
  (let [[min-x min-y min-z] (:min bb)
        [max-x max-y max-z] (:max bb)]
    [(/ (+ min-x max-x) 2.0)
     (/ (+ min-y max-y) 2.0)
     (/ (+ min-z max-z) 2.0)]))

(defn extent
  "Size of the AABB along each axis (max - min)."
  [^AABB bb]
  (v- (:max bb) (:min bb)))

(defn max-side
  "Largest side length of the AABB, used as size s in Barnes–Hut criterion."
  [^AABB bb]
  (let [[sx sy sz] (extent bb)]
    (max (Math/abs (double sx))
         (Math/abs (double sy))
         (Math/abs (double sz)))))

;; --- Octants ----------------------------------------------------------------

(defn octant
  "Classify point p into an octant of AABB bb relative to its center.
   Returns one of 8 keywords:
   :octant/ppp, :octant/ppm, :octant/pmp, :octant/pmm,
   :octant/mpp, :octant/mpm, :octant/mmp, :octant/mmm.
   Points exactly on a plane are treated as positive (>=) along that axis
   to keep classification deterministic."
  [^AABB bb [x y z]]
  (let [[cx cy cz] (center bb)
        px? (>= (double x) (double cx))
        py? (>= (double y) (double cy))
        pz? (>= (double z) (double cz))]
    (cond
      (and px? py? pz?) :octant/ppp
      (and px? py? (not pz?)) :octant/ppm
      (and px? (not py?) pz?) :octant/pmp
      (and px? (not py?) (not pz?)) :octant/pmm
      (and (not px?) py? pz?) :octant/mpp
      (and (not px?) py? (not pz?)) :octant/mpm
      (and (not px?) (not py?) pz?) :octant/mmp
      :else :octant/mmm)))

(defn child-aabb
  "Given parent AABB and an octant keyword, return the child's AABB."
  [^AABB bb oct]
  (let [[min-x min-y min-z] (:min bb)
        [max-x max-y max-z] (:max bb)
        [cx cy cz]          (center bb)]
    (case oct
      :octant/ppp (->AABB [cx cy cz] [max-x max-y max-z])
      :octant/ppm (->AABB [cx cy min-z] [max-x max-y cz])
      :octant/pmp (->AABB [cx min-y cz] [max-x cy max-z])
      :octant/pmm (->AABB [cx min-y min-z] [max-x cy cz])
      :octant/mpp (->AABB [min-x cy cz] [cx max-y max-z])
      :octant/mpm (->AABB [min-x cy min-z] [cx max-y cz])
      :octant/mmp (->AABB [min-x min-y cz] [cx cy max-z])
      :octant/mmm (->AABB [min-x min-y min-z] [cx cy cz]))))

;; --- Bodies -----------------------------------------------------------------

(defrecord Body
  [id        ;; entity id
   mass      ;; scalar mass
   radius    ;; scalar radius
   kind      ;; keyword, e.g. :body/star :body/planet :body/person
   position  ;; vec3
   velocity  ;; vec3
   ])

(defn ->body
  "Construct a Body from a map with keys:
   :id, :mass, :radius, :kind, :position, :velocity."
  [{:keys [id mass radius kind position velocity] :as m}]
  (when-not (and id mass radius kind position velocity)
    (throw (ex-info "Body requires :id, :mass, :radius, :kind, :position, :velocity"
                    {:kind ::invalid-body :body m})))
  (->Body id (double mass) (double radius) kind position velocity))
```

This gives you all the spatial primitives you need to express a Barnes–Hut octree in 3D as described in the literature.[^8_2][^8_1]

## `domain.gravity.barnes-hut`: octree + acceleration

This module implements a minimal Barnes–Hut octree in 3D: build from bodies → compute total mass and center of mass at each internal node → evaluate acceleration on a target body using the $s/d < \theta$ criterion.[^8_1][^8_2]

```clojure
(ns domain.gravity.barnes-hut
  "Barnes–Hut n-body gravity in 3D, using an octree over AABBs [web:46][web:49].
   - build-tree: bodies -> tree
   - acceleration: G θ tree body -> vec3 acceleration on body."
  (:require
    [shape.spatial :as sp]))

;; --- Node representation ----------------------------------------------------
;; We'll use simple maps rather than records while iterating:
;; Leaf:    {:type :leaf    :aabb bb :body body :mass m :com pos}
;; Internal {:type :internal :aabb bb :children [n0..n7 or nil] :mass m :com pos}
;; Empty:   nil

(defn leaf-node
  [^shape.spatial.AABB bb body]
  {:type :leaf
   :aabb bb
   :body body
   :mass (:mass body)
   :com  (:position body)})

(defn internal-node
  [^shape.spatial.AABB bb children mass com]
  {:type     :internal
   :aabb     bb
   :children children ;; vector of 8 child nodes or nils
   :mass     mass
   :com      com})

(defn internal-node? [node] (= (:type node) :internal))
(defn leaf-node?     [node] (= (:type node) :leaf))

;; --- Tree building ----------------------------------------------------------

(defn- bounding-aabb-for-bodies
  [bodies]
  (sp/aabb-from-points (map :position bodies)))

(def ^:private all-octants
  [:octant/ppp :octant/ppm :octant/pmp :octant/pmm
   :octant/mpp :octant/mpm :octant/mmp :octant/mmm])

(def ^:private octant-index
  "Stable index for each octant: 0..7."
  (zipmap all-octants (range 8)))

(defn- empty-internal
  "Create an empty internal node for AABB bb."
  [^shape.spatial.AABB bb]
  {:type     :internal
   :aabb     bb
   :children (vec (repeat 8 nil))
   :mass     0.0
   :com      (sp/vec3 0.0 0.0 0.0)})

(defn- insert-body-into-node
  "Insert body into node, subdividing as needed to maintain leaf capacity 1."
  [node body]
  (let [pos (:position body)]
    (cond
      (nil? node)
      (leaf-node (sp/aabb (:position body) (:position body)) body)

      (leaf-node? node)
      (let [^shape.spatial.AABB bb (:aabb node)
            ;; if we reached a degenerate AABB (min==max) we may need to expand it
            bb' bb
            internal (empty-internal bb')
            ;; reinsert existing leaf's body and new body
            b-old (:body node)
            internal1 (insert-body-into-node internal b-old)]
        (insert-body-into-node internal1 body))

      (internal-node? node)
      (let [^shape.spatial.AABB bb (:aabb node)
            oct   (sp/octant bb pos)
            idx   (octant-index oct)
            child (get (:children node) idx)
            child-bb (sp/child-aabb bb oct)
            child' (if (nil? child)
                     (leaf-node child-bb body)
                     (insert-body-into-node child body))]
        (assoc node :children (assoc (:children node) idx child')))

      :else
      (throw (ex-info "Unknown node type in insert-body-into-node"
                      {:node node})))))

(defn- accumulate-mass-com
  "Post-order traversal to compute mass and center of mass for internal nodes."
  [node]
  (cond
    (nil? node) nil

    (leaf-node? node) node

    (internal-node? node)
    (let [children' (mapv accumulate-mass-com (:children node))
          non-nil   (filter some? children')
          total-m   (reduce (fn [m ch] (+ m (:mass ch))) 0.0 non-nil)
          com       (if (zero? total-m)
                      (sp/center (:aabb node))
                      (reduce
                        (fn [acc ch]
                          (sp/v+ acc
                                 (sp/v* (:com ch) (/ (:mass ch) total-m))))
                        (sp/vec3 0.0 0.0 0.0)
                        non-nil))]
      (assoc node
             :children children'
             :mass     total-m
             :com      com))

    :else
    (throw (ex-info "Unknown node type in accumulate-mass-com" {:node node}))))

(defn build-tree
  "Build a Barnes–Hut octree from a collection of Body instances.
   Returns the root node, or nil if no bodies."
  [bodies]
  (let [bodies (vec bodies)]
    (when (seq bodies)
      (let [root-bb (bounding-aabb-for-bodies bodies)
            ;; start with empty internal node over root-bb, insert all bodies
            raw-root (reduce insert-body-into-node
                             (empty-internal root-bb)
                             bodies)]
        (accumulate-mass-com raw-root)))))

;; --- Acceleration computation ----------------------------------------------
;; Standard Barnes–Hut criterion: if s/d < θ use node's COM as single body,
;; otherwise recurse [web:46][web:49].

(def ^:private softening2
  "Gravitational softening length squared to avoid singularity."
  1.0e-6)

(defn- acceleration-from-mass
  "Acceleration on body due to mass M located at pos-m.
   G: gravitational constant
   body: Body
   M: scalar mass
   pos-m: vec3 position of mass"
  [G body M pos-m]
  (let [pos   (:position body)
        dir   (sp/v- pos-m pos)
        r2    (+ (sp/len2 dir) softening2)
        inv-r (Math/invSqrt r2) ;; 1/sqrt(r2) if you want, else 1.0/(Math/sqrt r2)
        inv-r3 (* inv-r inv-r inv-r)
        ;; a = G * M / r^2 * r̂ = G * M * r_vec / r^3
        scale (* G M inv-r3)]
    (sp/v* dir scale)))

(defn- node-acc
  "Recursive helper: acceleration contribution from node on body."
  [G θ node body]
  (cond
    (nil? node) (sp/vec3 0.0 0.0 0.0)

    (leaf-node? node)
    (let [b' (:body node)]
      (if (= (:id body) (:id b'))
        (sp/vec3 0.0 0.0 0.0)
        (let [pos-m (:position b')]
          (acceleration-from-mass G body (:mass b') pos-m))))

    (internal-node? node)
    (let [bb     (:aabb node)
          size   (sp/max-side bb)
          pos    (:position body)
          com    (:com node)
          d      (sp/dist pos com)]
      (if (or (zero? d) (< (/ size d) θ))
        ;; Approximate entire node as point mass at COM.
        (acceleration-from-mass G body (:mass node) com)
        ;; Need to recurse into children.
        (reduce
          (fn [acc ch]
            (sp/v+ acc (node-acc G θ ch body)))
          (sp/vec3 0.0 0.0 0.0)
          (:children node))))

    :else
    (throw (ex-info "Unknown node type in node-acc" {:node node}))))

(defn acceleration
  "Compute gravitational acceleration on `body` due to all other bodies
   represented by `tree` using Barnes–Hut approximation with opening
   angle θ [web:46][web:49]. G is the gravitational constant.

   Returns vec3 acceleration."
  [G θ tree body]
  (if (nil? tree)
    (sp/vec3 0.0 0.0 0.0)
    (node-acc (double G) (double θ) tree body)))
```

Notes and knobs:

- `softening2` can be tuned per scale of your world to avoid blow‑ups at very small separations, following standard n‑body practice.[^8_3][^8_4][^8_1]
- I left `Math/invSqrt` as a slot; you can either write your own fast $1/\sqrt{x}$ or just use `(/ 1.0 (Math/sqrt r2))`.
- `insert-body-into-node` currently uses a degenerate AABB when starting from `leaf-node` zero‑area boxes; you may later want to ensure the root AABB is padded slightly so subdividing actually progresses for co‑located or near‑co‑located bodies.

With these two namespaces in place:

- Your μ‑specs for spatial primitives and Barnes–Hut should be close to green or only need small tweaks.
- `domain.gravity.system` and `domain.gravity.parallel` can now be implemented in terms of these operations, using the ECS skeleton we already drafted and the standard Barnes–Hut scheduling pattern (build tree once, update bodies in parallel).[^8_4][^8_2][^8_1]
